Offline Service Worker (sw.js)Implement a lightweight Cache-First service worker:Pre-cached Assets: index.html, engine.js, manifest.webmanifest, CSS fonts/stylesheets, and icon assets.Cache Invalidation: Auto-bump cache version key on builds (undoku-v1.x.x).Registration Pipeline: Register the service worker in the main client HTML on window.load without blocking canvas initialization.// Service Worker Implementation (sw.js)
const CACHE_NAME = 'undoku-core-v1';
const ASSETS_TO_CACHE = [
  './',
  './index.html',
  './engine.js',
  './manifest.webmanifest'
];

self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => cache.addAll(ASSETS_TO_CACHE))
  );
  self.skipWaiting();
});

self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then((keys) =>
      Promise.all(
        keys.map((k) => {
          if (k !== CACHE_NAME) return caches.delete(k);
        })
      )
    )
  );
  self.clients.claim();
});

self.addEventListener('fetch', (event) => {
  event.respondWith(
    caches.match(event.request).then((cachedResponse) => {
      return cachedResponse || fetch(event.request);
    })
  );
});
